## Declaring Abstract classes...
```
// Abstract class...

abstract class Super{ // class declared with keyWord "Abstract"
  public Super(){System.out.println("Super constructor");}

  public void meth1(){
    System.out.println("Meth1 of super !");
  }
 abstract public  void meth2();// Method without curly paranthesis is asbstract method or we can declared it using abstract keyword too..
}

 // creating an sub class of abstract class...
 class sub extends Super{
  @Override  
  public void meth2(){
    System.out.println("Sub Method2");
  }
 }
 public class AbstractExample{
  public static void main(String args []){
    Super s = new sub();
    s.meth1();
    s.meth2();
  }
 }
```

## example of Abstrat class 
```
abstract class Shape{
  // abstract methods...
  abstract public  double area();
  abstract public  double perimeter();
}

// creating circle clas using Shape abstract class..
class circle extends  Shape{
  double readius;
  // constructor
  public circle(double r){
    this.readius = r;
  }
  // overriden methods ...
  @Override  
  public double area(){
    return  Math.PI*readius*readius;
  }
  // overriden method...
  @Override 
  public  double perimeter(){
    return  Math.PI*2*readius;
  }
  public  double circumference(){
    return perimeter();
  }
}

class rectangle extends  Shape{
  double length ,bredth;
  public  rectangle(double l, double b){
    this.length = l;
    this.bredth = b;
  }
  @Override 
  public  double area(){
    return length*bredth;
  }
  @Override 
  public  double perimeter(){
    return  2*(length+bredth);
  }
}

/**
 * AbstractExample
 */
public class AbstractExample {

  public  static void main(String args[]){
   Shape c = new circle(7.8);
   Shape r = new rectangle(75.25,80.5);
   System.out.println("The area of rectangle is "+r.area()+"\n perimeter of the rectangle is "+r.perimeter()+"\n area of circle is : "+c.area()+ "\n circumference "+c.perimeter());
  }
}
```
